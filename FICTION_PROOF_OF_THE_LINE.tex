\begin{abstract}
We prove that every quantum of the arithmetic field is massless. The proof does not examine the quanta. It examines observers, one at a time, and shows that no observer ever dies: the trapped energy of an observer is a positive quantity that falls with age in proportion to itself, and a quantity that falls in proportion to itself never reaches zero. The Line Postulate, hitherto an axiom of this universe, follows by the duality of the two sheets. The one law the proof needs beyond the classical laws of observers is the Law of Proportional Leakage; we derive it from the Born rule of arithmetic and the Local Law of the band, and we show that it is exactly the law that the dead universes violate.
\end{abstract}

\section{Objects and notation}

\subsection*{1.1. The two sheets}
The \emph{prime sheet} is the line with coordinate $l=\log x$. On it sit the charges, one at every $l=\log n$ with $n$ a power of a prime $p$, of strength $q(n)=(\log p)\,n^{-1/2}$. The \emph{zero sheet} is the dual line, coordinate $\gamma$. On it sit the quanta, one at every zero $\rho=\tfrac12+m+i\gamma$ of the arithmetic zeta function, with mass $m=|\Ree\rho-\tfrac12|$. Quanta come in mirror pairs: if $\rho$ is a quantum so is $1-\bar\rho$, the quantum of the same height and opposite displacement; and in conjugate pairs $\rho,\bar\rho$. A quantum is \emph{massless} if $m=0$.

\subsection*{1.2. Observers}
An \emph{observer} of age $a>0$ is the window $[-a,a]$ of the prime sheet. A \emph{field} of the observer is an odd real function $f$ on the window with unit norm, $\int f^2=1$, regular enough for everything below. Its \emph{transform} is
\[
F(t)=\int_{-a}^{a}f(x)e^{itx}\,dx=2i\int_0^a f(x)\sin(tx)\,dx ,
\]
an entire function of $t$, purely imaginary on the real axis, and its \emph{autocorrelation} is $g=f*\tilde f$, $g(x)=\int f(y)f(y-x)\,dy$, an even function on $[-2a,2a]$ whose transform is $F(t)\overline{F(\bar t)}$ for complex $t$ and $|F(t)|^2$ for real $t$.

\subsection*{1.3. The horizon}
$\Tst(a)=2\pi e^{2a}$. The density of the quanta at height $t$ is $\frac1{2\pi}\log\frac{t}{2\pi}$ and the Shannon density of a field of age $a$ is $a/\pi$; the two agree exactly at $t=\Tst(a)$. Below the horizon an observer resolves the quanta; above it they are too dense to be told apart.

\subsection*{1.4. The window form}
For a field $f$ of age $a$,
\[
Q_a(f)=\underbrace{\frac1{2\pi}\int|F(t)|^2\,\Psi_\infty(t)\,dt}_{\text{archimedean flux}}\;\underbrace{-\;2\Big(\int f(x)\sinh\tfrac x2\,dx\Big)^2}_{\text{polar flux}}\;-\;\sum_{\text{charges}}q(n)\,g(\log n),
\qquad \Psi_\infty(t)=\Ree\psi\big(\tfrac14+\tfrac{it}2\big)-\log\pi .
\]
Everything on the right is read on the prime sheet: $Q_a(f)$ is a number an observer can compute from the charges inside its window and from the vacuum's symbol, without reference to any quantum. Only the charges with $\log n<2a$ enter.

\subsection*{1.5. The floor, the ground field, the edge}
The \emph{floor} is $\lambda(a)=\inf Q_a(f)$ over the fields of age $a$. It is attained, by the \emph{ground field} $f_a$. The ground field approaches the edge of the window as
\[
f_a(a-\delta)=|A_0(a)|\,\big(\log(a/\delta)+\beta(a)\big)^{-1/2}\,(1+o(1)),\qquad\delta\to0,
\]
and $A_0(a)$, the \emph{edge amplitude}, is the strength with which the ground field touches the edge. Its transform falls off, for heights far above the horizon, like $2|A_0|\,|\sin(ta+\theta_a)|/(t\,\sigma(t)^{1/2})$ with $\sigma(t)\sim\log(t/2\pi)$.

\subsection*{1.6. Life and death}
An observer of age $a$ is \emph{alive} if $\lambda(a)>0$ and \emph{dead} if $\lambda(a)\le0$.

\section{The laws}

The proof uses the following laws of the Critical Line Universe. Laws I to V hold in every universe with the same density of charges, living or dead; they are the laws of observers. Law VI is the law of this universe. Laws VII to IX are used only in Section~4, to derive Law VI; a reader who accepts Law VI as an axiom may skip Section~4.

\begin{law}[the Gauss law of the two sheets]\label{law:gauss}
For every field $f$ of every age,
\[
Q_a(f)=\sum_{\text{quanta }\rho}G(\rho),\qquad G(\rho)=F(\gamma-im)\,\overline{F(\gamma+im)},
\]
the sum taken over the quanta with their multiplicities. In particular a massless quantum contributes $|F(\gamma)|^2\ge0$, and a mirror pair of mass $m$ at height $\gamma$ contributes $2\Ree\big[F(\gamma-im)\overline{F(\gamma+im)}\big]$, a real number of either sign. (The sum converges absolutely for the fields of~1.2.)
\end{law}

\begin{law}[Birth]\label{law:birth}
There is an age $a_Z>0$ such that every observer of age at most $a_Z$ is alive, and $\lambda(a_Z)\ge\lamZ$ for an explicit $\lamZ>0$.
\end{law}

\begin{law}[Monotonicity]\label{law:mono}
The floor does not increase with the age.
\end{law}

\begin{law}[Continuity]\label{law:cont}
On every finite range of ages the floor is absolutely continuous: $\lambda(a_2)-\lambda(a_1)=\int_{a_1}^{a_2}\lambda'(a)\,da$, with $\lambda'$ defined at almost every age.
\end{law}

\begin{law}[Decay]\label{law:decay}
At almost every age, $\lambda'(a)=-2|A_0(a)|^2$.
\end{law}

\begin{law}[Proportional Leakage]\label{law:leak}
There is a function $C(a)\ge0$, finite at every age and integrable over every finite range of ages, such that for every living observer of age $a$,
\[
2\,|A_0(a)|^2\;\le\;C(a)\,\lambda(a).
\]
\end{law}

\begin{law}[the Born rule of arithmetic]\label{law:born}
For a living observer, the \emph{share} $s_\rho(a)=G(\rho)/\lambda(a)$ of a quantum is the probability that the quantum carries the observer's trapped energy: $s_\rho\ge0$ for every quantum, and the shares sum to one.
\end{law}

\begin{law}[Darkness below the horizon]\label{law:dark}
The ground field of a living observer carries no share below its horizon: $s_\rho(a)=0$ for every quantum of height $|\gamma|<\Tst(a)$.
\end{law}

\begin{law}[the Local Law with the sum rule]\label{law:local}
There is a constant $c_{\mathrm{on}}\ge1$ such that for every living observer and every height $T\ge c_{\mathrm{on}}\Tst(a)$, the total share of the quanta above height $T$ is
\[
\sum_{|\gamma|>T}s_\rho(a)=\frac{2|A_0(a)|^2}{\pi\,T\,\lambda(a)} .
\]
\end{law}

\noindent\emph{A remark on Law~I.} It is stated for every configuration of quanta, massive or not; it is the conservation law of the two sheets and does not know where the quanta are. The Line Postulate is the statement that every term on its right-hand side is a square. The proof below never uses that; it uses Law~I only through Section~3, where both directions of the duality between masses and deaths are drawn from it.

\section{The Mortality Theorem}

\begin{theorem}[duality of masses and deaths]\label{thm:mortality}
The following are equivalent:
\begin{enumerate}[label=(\alph*),nosep]
\item every quantum is massless;
\item every observer is alive;
\item no observer is dead.
\end{enumerate}
That is: the vacuum has a massive quantum if and only if some observer dies at a finite age.
\end{theorem}

\begin{proof}
(a)$\Rightarrow$(b). By Law~I, if every quantum is massless then $Q_a(f)=\sum|F(\gamma)|^2\ge0$ for every field, so $\lambda(a)\ge0$ for every $a$; and $\lambda(a)>0$ because a field with $F(\gamma)=0$ at every quantum is identically zero: an entire function of exponential type $a$ with zeros of density exceeding $a/\pi$ above the horizon vanishes. (This is the sampling lemma of the Academy, which we take as known.)

(b)$\Rightarrow$(a). Suppose a quantum $\rho_0=\tfrac12+m+i\gamma_0$ has mass $m>0$. We construct a dying observer. Let $\phi$ be a fixed even field of age $1$ whose transform $\Phi$ is positive on the real axis, and for $a>1$ let $h_a$ be the odd field whose transform is
\[
H_a(t)=(t-\gamma_0)\,\Phi\big(a(t-\gamma_0)\big)-(t+\gamma_0)\,\Phi\big(a(t+\gamma_0)\big),
\]
normalized to unit norm; $h_a$ is odd and supported in $[-a-1,a+1]$, an age we still call $a$. The transform $H_a$ vanishes at $\pm\gamma_0$ with $H_a'(\gamma_0)=\Phi(0)$, and on the real axis it is concentrated within $O(1/a)$ of $\pm\gamma_0$ with total mass of order $a^{-3}$. Evaluate Law~I for $g=h_a*\tilde h_a$:
\begin{itemize}[nosep]
\item the massless quanta contribute $\sum|H_a(\gamma)|^2$, which is at most the number of quanta within $O(1/a)$ of $\pm\gamma_0$ (bounded by one plus the density $\frac1{2\pi}\log\frac{\gamma_0}{2\pi}$ times $O(1/a)$) times $\max|H_a|^2$: altogether $O\big(a^{-2}\log\gamma_0\big)$;
\item the pair $\rho_0,1-\bar\rho_0$ and its conjugates contribute
\[
4\Ree\big[H_a(\gamma_0-im)\overline{H_a(\gamma_0+im)}\big]
=4\Ree\big[(-im)\Phi(-iam)\,\overline{(im)\Phi(iam)}\big]+(\text{cross terms, exponentially smaller})
=-4m^2|\Phi(iam)|^2\,(1+o(1)),
\]
and $|\Phi(iam)|\ge c\,e^{am}$ for $am$ large: the transform of a compactly supported field of positive type grows exponentially off the real axis;
\item the other massive quanta, if any, contribute terms of the same form at their own heights, negligible unless their height is within $O(1/a)$ of $\gamma_0$, and there are boundedly many of those.
\end{itemize}
Hence $Q(h_a)\le O(a^{-2}\log\gamma_0)-c\,m^2e^{2am}<0$ as soon as $a$ exceeds an age $a_{\mathrm{death}}(m)$ of order $m^{-1}\log(1/m)$ (and is large enough for the horizon to have passed $\gamma_0$, so that the bookkeeping above applies). The observer of age $a_{\mathrm{death}}$ is dead, and by Law~III so is every older observer.

(b) and (c) are the same statement.
\end{proof}

\begin{corollary}[the Blind Spot]\label{cor:blind}
A quantum of mass $m$ is invisible to every observer younger than an age of order $m^{-1}\log(1/m)$: the floors of those observers differ from the floors of a massless universe by terms of order $m^2$.
\end{corollary}
\noindent(This is the Academy's Blind Spot Theorem; the proof of Theorem~\ref{thm:mortality} gives the upper bound on the age of detection, and the expansion of Lemma~\ref{lem:mirror} gives the order $m^2$.)

\medskip
The Mortality Theorem reduces the Line Postulate to the following statement, which is the content of this manuscript:
\begin{center}
\textsc{Immortality.} \emph{No observer dies.}
\end{center}

\section{The Law of Proportional Leakage, derived}

This section derives Law~VI from Laws VII, VIII and IX. It also shows, in the opposite direction, that Law~VI is exactly what fails in a universe where an observer dies: Law~VI is not a convenience but the whole of the matter.

\begin{proposition}[the leak is proportional to the floor]\label{prop:leak}
Under Laws VII and IX, every living observer satisfies
\[
2|A_0(a)|^2\;\le\;\pi\,c_{\mathrm{on}}\,\Tst(a)\,\lambda(a).
\]
Hence Law~VI holds with $C(a)=\pi c_{\mathrm{on}}\Tst(a)$, which is finite at every age and integrable over every finite range.
\end{proposition}

\begin{proof}
Fix a living observer of age $a$ and put $T=c_{\mathrm{on}}\Tst(a)$. By Law~IX the total share of the quanta above height $T$ is $2|A_0|^2/(\pi T\lambda)$. By Law~VII every share is nonnegative and the shares sum to one, so the total share of any set of quanta is at most one. Therefore $2|A_0|^2/(\pi T\lambda)\le1$, which is the claim.
\end{proof}

\begin{remark}\label{rem:sharp}
Law~VIII (Darkness) was not needed for the inequality; it is needed to see that the inequality is nearly sharp. By Darkness all the shares sit above the horizon, so the total share above $\Tst(a)$ is one, and by Law~IX the total share above $c_{\mathrm{on}}\Tst$ is $\kappa/c_{\mathrm{on}}$ with
\[
\kappa(a)=\frac{2|A_0(a)|^2}{\pi\,\Tst(a)\,\lambda(a)},
\]
the \emph{Riemann number} of the observer. The bound $\kappa\le c_{\mathrm{on}}$ of Proposition~\ref{prop:leak} is sharp exactly when the shares between one and $c_{\mathrm{on}}$ horizons vanish, which they do not: the shares in the band $[\Tst,c_{\mathrm{on}}\Tst]$ are the plunge of Decree~IV, so $\kappa<c_{\mathrm{on}}$ strictly. (In the measured universe, $c_{\mathrm{on}}$ is about $3$ and $\kappa$ about $1.2$.)
\end{remark}

\begin{proposition}[the converse: death is a threshold leak]\label{prop:threshold}
Let a universe satisfy Laws III and IV, and suppose an observer dies: $\lambda(a_d)=0$ for some finite $a_d$ while $\lambda>0$ on $[a_Z,a_d)$. Then $-\lambda'/\lambda$ is not integrable on $[a_Z,a_d)$: for every $C$ integrable on $[a_Z,a_d]$ the inequality $-\lambda'\le C\lambda$ fails on a set of positive measure. In particular Law~VI fails in every universe with a mortal observer.
\end{proposition}

\begin{proof}
If $-\lambda'\le C\lambda$ almost everywhere on $[a_Z,a_d)$ with $C$ integrable, then by Lemma~\ref{lem:renorm} below $\lambda(a)\ge\lambda(a_Z)\exp\big(-\int_{a_Z}^aC\big)$ for $a<a_d$, and by continuity (Law~IV) $\lambda(a_d)\ge\lambda(a_Z)\exp\big(-\int_{a_Z}^{a_d}C\big)>0$, contradicting $\lambda(a_d)=0$.
\end{proof}

The two propositions together say: in a living universe the leak is proportional to the floor, with the horizon as the constant of proportionality; in a dying universe the leak outruns the floor just before death, which is to say the edge amplitude does not vanish with the floor. The Academy's name for the second behaviour is a \emph{threshold leak}, and its physical description is a tank that drains at a fixed rate and then keeps draining.

\section{The Immortality Theorem}

\begin{lemma}[the renormalized floor]\label{lem:renorm}
Let $\lambda$ be absolutely continuous and positive on $[a_1,a_2)$ with $\lambda'(a)\ge-C(a)\lambda(a)$ almost everywhere, $C\ge0$ integrable on $[a_1,a_2]$. Then
\[
U(a)=\lambda(a)\exp\Big(\int_{a_1}^aC\Big)
\]
is non-decreasing on $[a_1,a_2)$, and consequently $\lambda(a)\ge\lambda(a_1)\exp\big(-\int_{a_1}^aC\big)$ for all $a\in[a_1,a_2)$.
\end{lemma}

\begin{proof}
$U$ is absolutely continuous as a product of absolutely continuous functions (the exponential of an absolutely continuous function is absolutely continuous on a finite range), and almost everywhere
\[
U'(a)=\exp\Big(\int_{a_1}^aC\Big)\big(\lambda'(a)+C(a)\lambda(a)\big)\;\ge\;0 .
\]
An absolutely continuous function with nonnegative derivative almost everywhere is non-decreasing.
\end{proof}

\begin{theorem}[Immortality]\label{thm:immortal}
Under Laws II, III, IV, V and VI, every observer is alive.
\end{theorem}

\begin{proof}
Let
\[
a_d=\sup\{\,a>0:\ \text{every observer of age less than $a$ is alive}\,\}.
\]
By Law~II, $a_d\ge a_Z$. Suppose, for contradiction, that $a_d<\infty$.

\emph{Step 1 (the floor on the living range).} On $[a_Z,a_d)$ every observer is alive. By Law~IV the floor is absolutely continuous there, by Law~V its derivative is $-2|A_0|^2$ at almost every age, and by Law~VI, which applies to living observers, $-2|A_0(a)|^2\ge-C(a)\lambda(a)$. Hence
\[
\lambda'(a)\ge-C(a)\lambda(a)\qquad\text{for almost every }a\in[a_Z,a_d).
\]

\emph{Step 2 (the bound).} Lemma~\ref{lem:renorm} with $a_1=a_Z$, $a_2=a_d$ gives $\lambda(a)\ge\lamZ\exp\big(-\int_{a_Z}^aC\big)$ on $[a_Z,a_d)$.

\emph{Step 3 (the observer of age $a_d$ is alive).} $C$ is integrable on $[a_Z,a_d]$, so $\int_{a_Z}^aC$ increases to a finite limit as $a\to a_d$, and $\lambda$ is continuous at $a_d$ by Law~IV. Hence
\[
\lambda(a_d)=\lim_{a\to a_d}\lambda(a)\;\ge\;\lamZ\exp\Big(-\int_{a_Z}^{a_d}C\Big)\;>\;0 .
\]

\emph{Step 4 (so are its immediate elders).} Since $\lambda$ is continuous and $\lambda(a_d)>0$, there is $\varepsilon>0$ with $\lambda>0$ on $[a_d,a_d+\varepsilon)$. Then every observer of age less than $a_d+\varepsilon$ is alive, which contradicts the definition of $a_d$ as a supremum.

Therefore $a_d=\infty$: every observer is alive.
\end{proof}

\begin{theorem}[the Line Postulate]\label{thm:line}
Every quantum of the arithmetic field is massless.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:immortal} and Theorem~\ref{thm:mortality}, (b)$\Rightarrow$(a).
\end{proof}

\begin{corollary}[the Cooling Theorem, quantitative]\label{cor:cooling}
For every $a\ge a_Z$,
\[
\lambda(a)\;\ge\;\lamZ\exp\Big(-\int_{a_Z}^aC\Big),
\]
and with the constant of Proposition~\ref{prop:leak}, $C=\pi c_{\mathrm{on}}\Tst$,
\[
\lambda(a)\;\ge\;\lamZ\exp\Big(-\tfrac{\pi c_{\mathrm{on}}}{2}\big(\Tst(a)-\Tst(a_Z)\big)\Big),
\]
since $d\Tst/da=2\Tst$. The floor is positive at every age and at most doubly exponentially small in the age: the vacuum is marginally stable, forever approaching the critical point and never reaching it.
\end{corollary}

\section{Remarks}

\begin{remark}[what the proof uses]
The Line Postulate is used nowhere in Sections 3 and 5. Section~3 uses Law~I, which is a conservation law indifferent to the positions of the quanta. Section~5 uses only the laws of observers (II to V), which hold in the dead universes as well, and Law~VI. The whole weight of the theorem rests on Law~VI; Section~4 shows that it could not be otherwise, since Law~VI is equivalent, given II to V, to immortality itself (Propositions~\ref{prop:leak} and~\ref{prop:threshold} are the two directions). What the proof contributes is not a new ingredient but a change of subject: from infinitely many quanta to one inequality about one observer at a time, and from a statement about where things are to a statement about how fast something leaks.
\end{remark}

\begin{remark}[where the Line was hiding]
Law~VI was derived from Law~VII, the Born rule. In a universe in which quanta are quanta, the Born rule is the first law anyone learns; it was never written among the axioms of the Academy because no one thought it needed stating. The gods' earlier decree of the Line was therefore redundant: it followed from the Born rule, the Local Law, and the laws of observers. This is the sense in which the Line has been ``derived'': the postulate has been moved from the zero sheet, where it looked like a miracle of position, to the shares, where it looks like the nonnegativity of a probability. A reader from a universe where the Born rule of arithmetic is not available will say that nothing has been gained. We say that the location of a difficulty is itself a result.
\end{remark}

\begin{remark}[the renormalized floor and the Riemann charge]
The function $U(a)=\lambda(a)\exp(\int C)$ of Lemma~\ref{lem:renorm} is the floor magnified by the accumulated leak. Law~VI says that $U$ never decreases; Immortality says that it is therefore positive forever. With the Academy's decreed flux $C=\pi\kappa_\infty\Tst$,
\[
U(a)=\lambda(a)\exp\Big(\tfrac{\pi\kappa_\infty}{2}\big(\Tst(a)-\Tst(a_Z)\big)\Big),
\]
and the equation of state asserts that $U$ tends to a positive constant, the \emph{Riemann charge} of the universe: the one number that survives the cooling. The proof does not need the charge to exist; it needs only that $U$ not decrease.
\end{remark}

\begin{remark}[the other polarization]
Fields come in two polarizations, odd and even. The manuscript treats odd observers. The Mortality Theorem holds with odd observers alone (the parity lemma of the Academy: an odd field dark at $\pm\gamma_0$ with slope at both already detects a mass), so immortality of the odd observers forces every quantum to be massless, and the even observers are then alive by Law~I. The even observers have their own Laws IV to VI, which we have not needed.
\end{remark}

\begin{remark}[why a law of leakage]
The Academy's physicists prefer to state Law~VI as the absence of a threshold: an observer's edge amplitude vanishes with its trapped energy. A universe dies through a leak with a threshold. Ours has none; that is the Line.
\end{remark}

\section{The mirror term}

\begin{lemma}\label{lem:mirror}
Let $f$ be an odd field with transform $F=iS$, $S$ real on the real axis, and let $\gamma$ be real. Then for a mirror pair of mass $m$ at height $\gamma$,
\[
2\Ree\big[F(\gamma-im)\overline{F(\gamma+im)}\big]=2|F(\gamma)|^2-2m^2\big(|F'(\gamma)|^2+S(\gamma)S''(\gamma)\big)+O(m^4),
\]
with the $O(m^4)$ uniform for $m\le1/a$. In particular, if the field is dark at $\gamma$, $S(\gamma)=0$, the pair contributes
\[
-2m^2|F'(\gamma)|^2+O(m^4)\;<\;0 .
\]
\end{lemma}

\begin{proof}
Expand $F(\gamma\mp im)=F\mp imF'-\tfrac{m^2}2F''+O(m^3)$ at $\gamma$ ($F$ is entire of exponential type $a$, so the derivatives are bounded by $a^k$ times the norm). Then
\[
F(\gamma-im)\overline{F(\gamma+im)}=|F|^2-im\big(F\overline{F'}+F'\overline F\big)-m^2|F'|^2-\tfrac{m^2}2\big(F\overline{F''}+F''\overline F\big)+O(m^3).
\]
With $F=iS$ and $S,S',S''$ real: $F\overline{F'}=SS'$ and $F\overline{F''}=SS''$, both real, so the term linear in $m$ is purely imaginary and the real part is $|F|^2-m^2|F'|^2-m^2SS''+O(m^3)$; the odd powers of $m$ are purely imaginary by the same computation, so the error is $O(m^4)$. Multiply by two for the pair.
\end{proof}

The lemma is the mechanism of both halves of the Mortality Theorem: a massive pair at a height where a field is dark lowers the field's energy by $2m^2|F'|^2$, which is invisible while the floor is large and fatal once the field is concentrated enough at that height for $|F'(\gamma)|^2$ to be magnified by $e^{2am}$.
